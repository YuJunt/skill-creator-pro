🚀 快速开始

安装：下载Release中的zip包，解压到 workspace/.user_skills/ 目录

# 新建技能
python3 scripts/create_skill.py create /path/to/new-skill

# 优化现有技能
python3 scripts/create_skill.py optimize /path/to/existing-skill

# 深度评审
python3 scripts/create_skill.py review /path/to/skill

# 端到端测试
python3 scripts/create_skill.py test /path/to/skill

# 规范校验
python3 scripts/validate_skill.py /path/to/skill

# 安全扫描
python3 scripts/security_scan.py /path/to/skill
